-- WINDOW DRAGGING & RESIZING IMPLEMENTATION
--------------------------------------------------------------------------------
local TopBar = Instance.new("Frame")
TopBar.Name = "TopBar"
TopBar.Size = UDim2.new(1, 0, 0, 38)
TopBar.BackgroundTransparency = 1
TopBar.Parent = MainFrame

-- Dragging logic
local dragging, dragInput, dragStart, startPos
TopBar.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = input.Position
		startPos = MainFrame.Position
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then dragging = false end
		end)
	end
end)

TopBar.InputChanged:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
		dragInput = input
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if input == dragInput and dragging then
		local delta = input.Position - dragStart
		MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
	end
end)

-- Resize Grip Handler (Bottom-Right Corner)
local ResizeHandle = Instance.new("TextButton")
ResizeHandle.Name = "ResizeHandle"
ResizeHandle.Size = UDim2.new(0, 20, 0, 20)
ResizeHandle.Position = UDim2.new(1, -20, 1, -20)
ResizeHandle.BackgroundTransparency = 1
ResizeHandle.Text = "◢"
ResizeHandle.TextColor3 = Color3.fromRGB(80, 90, 105)
ResizeHandle.TextSize = 12
ResizeHandle.ZIndex = 20
ResizeHandle.Parent = MainFrame

local resizing = false
local resizeStart, startSize

ResizeHandle.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		resizing = true
		resizeStart = input.Position
		startSize = MainFrame.AbsoluteSize
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then resizing = false end
		end)
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if resizing and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local delta = input.Position - resizeStart
		local newX = math.clamp(startSize.X + delta.X, MainConstraint.MinSize.X, MainConstraint.MaxSize.X)
		local newY = math.clamp(startSize.Y + delta.Y, MainConstraint.MinSize.Y, MainConstraint.MaxSize.Y)
		MainFrame.Size = UDim2.new(0, newX, 0, newY)
	end
end)

local function makeDot(color, callback)
	local dot = Instance.new("TextButton")
	dot.Size = UDim2.new(0, 12, 0, 12)
	dot.BackgroundColor3 = color
	dot.Text = ""
	dot.AutoButtonColor = false
	dot.Parent = WindowControls
	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(1, 0)
	corner.Parent = dot
	if callback then
		dot.MouseButton1Click:Connect(callback)
	end
	return dot
end

-- Close Button (Red)
makeDot(Color3.fromRGB(235, 90, 90), function()
	ScreenGui:Destroy()
end)

-- Minimize Button (Yellow) - Toggles body visibility and collapses window height
local minimized = false
local previousSize = MainFrame.Size
makeDot(Color3.fromRGB(240, 190, 70), function()
	minimized = not minimized
	if minimized then
		previousSize = MainFrame.Size
		SideBar.Visible = false
		ContentPanel.Visible = false
		ResizeHandle.Visible = false
		MainFrame.Size = UDim2.new(previousSize.X.Scale, previousSize.X.Offset, 0, 38)
	else
		SideBar.Visible = true
		ContentPanel.Visible = true
		ResizeHandle.Visible = true
		MainFrame.Size = previousSize
	end
end)

-- Maximize Button (Green) - Toggles between maximized size and normal size
local maximized = false
local normalSize = MainFrame.Size
makeDot(Color3.fromRGB(80, 200, 120), function()
	maximized = not maximized
	if maximized then
		normalSize = MainFrame.Size
		MainFrame.Size = UDim2.new(0.95, 0, 0.9, 0)
		MainFrame.Position = UDim2.new(0.5, 0, 0.5, 0)
	else
		MainFrame.Size = normalSize
		MainFrame.Position = UDim2.new(0.5, 0, 0.5, 0)
	end
end)
